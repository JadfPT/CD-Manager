alter table public.artists
add column if not exists image_url text;


-- Bucket
insert into storage.buckets (id, name, public)
values ('artist', 'artist', true)
on conflict (id) do nothing;

-- Leitura pública
create policy "Artist public read"
on storage.objects
for select
to public
using (bucket_id = 'artist');

-- Insert na própria pasta (auth.uid()/...)
create policy "Artist auth insert own folder"
on storage.objects
for insert
to authenticated
with check (
  bucket_id = 'artist'
  and (storage.foldername(name))[1] = auth.uid()::text
);

-- Update na própria pasta
create policy "Artist auth update own folder"
on storage.objects
for update
to authenticated
using (
  bucket_id = 'artist'
  and (storage.foldername(name))[1] = auth.uid()::text
)
with check (
  bucket_id = 'artist'
  and (storage.foldername(name))[1] = auth.uid()::text
);

-- Delete na própria pasta
create policy "Artist auth delete own folder"
on storage.objects
for delete
to authenticated
using (
  bucket_id = 'artist'
  and (storage.foldername(name))[1] = auth.uid()::text
);
