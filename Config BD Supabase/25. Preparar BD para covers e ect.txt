begin;

-- tracks reais dos itens
create table if not exists public.item_tracks (
  id bigint generated always as identity primary key,
  item_type text not null check (item_type in ('cd', 'vinyl')),
  item_id bigint not null,
  disc_number integer not null default 1,
  position integer not null,
  title text not null,
  duration_ms integer,
  source text,
  source_id text,
  created_at timestamptz not null default timezone('utc', now()),
  unique (item_type, item_id, disc_number, position)
);

alter table public.item_tracks enable row level security;

drop policy if exists "item_tracks_select_authenticated" on public.item_tracks;
drop policy if exists "item_tracks_insert_admin" on public.item_tracks;
drop policy if exists "item_tracks_update_admin" on public.item_tracks;
drop policy if exists "item_tracks_delete_admin" on public.item_tracks;

create policy "item_tracks_select_authenticated"
on public.item_tracks
for select
to authenticated
using (true);

create policy "item_tracks_insert_admin"
on public.item_tracks
for insert
to authenticated
with check (public.current_user_is_admin());

create policy "item_tracks_update_admin"
on public.item_tracks
for update
to authenticated
using (public.current_user_is_admin())
with check (public.current_user_is_admin());

create policy "item_tracks_delete_admin"
on public.item_tracks
for delete
to authenticated
using (public.current_user_is_admin());

grant select, insert, update, delete on public.item_tracks to authenticated;

-- sugestões de tracklist
create table if not exists public.track_suggestions (
  id bigint generated always as identity primary key,
  item_type text not null check (item_type in ('cd', 'vinyl')),
  item_id bigint not null,
  source text not null,
  source_id text,
  confidence numeric not null default 0,
  tracks jsonb not null,
  status text not null default 'pending'
    check (status in ('pending', 'approved', 'rejected', 'applied')),
  created_at timestamptz not null default timezone('utc', now())
);

alter table public.track_suggestions enable row level security;

drop policy if exists "track_suggestions_select_admin" on public.track_suggestions;
drop policy if exists "track_suggestions_insert_admin" on public.track_suggestions;
drop policy if exists "track_suggestions_update_admin" on public.track_suggestions;

create policy "track_suggestions_select_admin"
on public.track_suggestions
for select
to authenticated
using (public.current_user_is_admin());

create policy "track_suggestions_insert_admin"
on public.track_suggestions
for insert
to authenticated
with check (public.current_user_is_admin());

create policy "track_suggestions_update_admin"
on public.track_suggestions
for update
to authenticated
using (public.current_user_is_admin())
with check (public.current_user_is_admin());

grant select, insert, update on public.track_suggestions to authenticated;

-- campos extra nas sugestões de capa
alter table public.cover_suggestions
add column if not exists local_storage_path text,
add column if not exists local_public_url text,
add column if not exists release_title text,
add column if not exists artist_name text,
add column if not exists matched_track_count integer,
add column if not exists source_url text;

-- metadata nos CDs/vinis
alter table public.cd_albums
add column if not exists metadata_source text,
add column if not exists metadata_source_id text,
add column if not exists cover_source text,
add column if not exists cover_confidence numeric;

alter table public.vinyl_albums
add column if not exists metadata_source text,
add column if not exists metadata_source_id text,
add column if not exists cover_source text,
add column if not exists cover_confidence numeric;

commit;
